   292→   292→function DescriptionForm({
   293→   293→  mon,
   294→   294→  nickname,
   295→   295→  setNickname,
   296→   296→}: {
   297→   297→  mon: Mon;
   298→   298→  nickname: string;
   299→   299→  setNickname: (v: string) => void;
   300→   300→}) {
   301→   301→  const [text, setText] = useState("");
   302→   302→  const [state, setState] = useState<"idle" | "sending" | "done">("idle");
   303→   303→  const [error, setError] = useState<string | null>(null);
   304→   304→
   305→   305→  const clean = passesQuickFilter(text);
   306→   306→  const tooShort = text.trim().length < 12;
   307→   307→
   308→   308→  async function submit() {
   309→   309→    setError(null);
   310→   310→    if (!supabaseConfigured) {
   311→   311→      setError("Connection unavailable — try again soon.");
   312→   312→      return;
   313→   313→    }
   314→   314→    if (!clean) {
   315→   315→      setError("Your text didn't pass the safety filter — please rephrase.");
   316→   316→      return;
   317→   317→    }
   318→   318→    setState("sending");
   319→   319→    try {
   320→   320→      const { error: err } = await supabase.from("proposals").insert({
   321→   321→        mon_id: mon.id,
   322→   322→        kind: "description",
   323→   323→        content: text.trim().slice(0, 500),
   324→   324→        submitted_by: canonicalize(nickname) || "anonymous",
   325→   325→      });
   326→   326→      if (err) {
   327→   327→        const msg = /safety filter/i.test(err.message)
   328→   328→          ? "Blocked by the safety filter — please rephrase kindly."
   329→   329→          : err.message;
   330→   330→        setError(msg);
   331→   331→        setState("idle");
   332→   332→        return;
   333→   333→      }
   334→   334→      setState("done");
   335→   335→    } catch {
   336→   336→      setError("Network hiccup — try again.");
   337→   337→      setState("idle");
   338→   338→    }
   339→   339→  }
   340→   340→
   341→   341→  if (state === "done") {
   342→   342→    return (
   343→   343→      <div className="space-y-3">
   344→   344→        <Alert className="border-pokedex-cyan/40 bg-pokedex-cyan/10">
   345→   345→          <ShieldCheck className="h-4 w-4 text-pokedex-cyan" />
   346→   346→          <AlertDescription className="font-soft text-sm">
   347→   347→            Submitted! The channel team reviews every entry before it appears — thanks, researcher.
   348→   348→          </AlertDescription>
   349→   349→        </Alert>
   350→   350→        <Button variant="secondary" className="font-soft rounded-full font-bold" onClick={() => setState("idle")}>
   351→   351→          Propose another
   352→   352→        </Button>
   353→   353→      </div>
   354→   354→    );
   355→   355→  }
   356→   356→
   357→   357→  return (
   358→   358→    <div className="space-y-4">
   359→   359→      <NicknameField nickname={nickname} setNickname={setNickname} />
   360→   360→      <div className="space-y-1.5">
   361→   361→        <label htmlFor="lp-desc" className="font-soft text-sm font-bold text-muted-foreground">
   362→   362→          What is {displayName(mon.name)}? (max {MAX_DESCRIPTION} chars)
   363→   363→        </label>
   364→   364→        <Textarea
   365→   365→          id="lp-desc"
   366→   366→          value={text}
   367→   367→          maxLength={MAX_DESCRIPTION}
   368→   368→          onChange={(e) => setText(e.target.value)}
   369→   369→          placeholder="A gremlin of pure chaos that only appears when someone drops a combo..."
   370→   370→          rows={4}
   371→   371→          className="resize-none rounded-2xl bg-secondary/70 font-semibold"
   372→   372→        />
   373→   373→        <div className="flex items-center gap-3">
   374→   374→          <Progress value={(text.length / MAX_DESCRIPTION) * 100} className="h-1.5" />
   375→   375→          <span className="font-soft shrink-0 text-xs text-muted-foreground">
   376→   376→            {text.length}/{MAX_DESCRIPTION}
   377→   377→          </span>
   378→   378→        </div>
   379→   379→        {!clean && (
   380→   380→          <p className="font-soft text-xs text-destructive">
   381→   381→            That wording will be blocked by the safety filter — please rephrase kindly.
   382→   382→          </p>
   383→   383→        )}
   384→   384→      </div>
   385→   385→      {error && <p className="font-soft text-sm text-destructive">{error}</p>}
   386→   386→      <Button
   387→   387→        onClick={submit}
   388→   388→        disabled={state === "sending" || tooShort || !clean}
   389→   389→        className="font-display w-full rounded-full text-sm font-bold shadow-[0_6px_18px_rgba(240,107,168,0.35)] hover:bg-pokedex-dark-red"
   390→   390→      >
   391→   391→        {state === "sending" ? (
   392→   392→          <>
   393→   393→            <Loader2 className="mr-2 h-4 w-4 animate-spin" /> Sending…
   394→   394→          </>
   395→   395→        ) : (
   396→   396→          <>
   397→   397→            <Sparkles className="mr-2 h-4 w-4" /> Submit for review
   398→   398→          </>
   399→   399→        )}
   400→   400→      </Button>
   401→   401→      <p className="flex items-center gap-1.5 text-xs text-muted-foreground">
   402→   402→        <ShieldCheck className="h-3.5 w-3.5 text-pokedex-cyan" />
   403→   403→        Safety: filtered for bad words + human-approved by the channel team before publishing.
   404→   404→      </p>
   405→   405→    </div>
   406→   406→  );
   407→   407→}

   409→   409→function ImageForm({

   412→   412→  setNickname,

   415→   415→  nickname: string;

   417→   417→}) {
   418→   418→  const [file, setFile] = useState<File | null>(null);
   419→   419→  const [preview, setPreview] = useState<string | null>(null);
   420→   420→  // Artwork source: a hand-picked file, or one found via the GIPHY picker.
   421→   421→  // The GIPHY option only renders when a free API key is configured.
   422→   422→  const [source, setSource] = useState<"upload" | "giphy">("upload");
   423→   423→  const [gifPick, setGifPick] = useState<GiphyGif | null>(null);
   424→   424→  const [state, setState] = useState<"idle" | "sending" | "done">("idle");
   425→   425→  const [progressMsg, setProgressMsg] = useState("");
   426→   426→  const [error, setError] = useState<string | null>(null);
   427→   427→  const inputRef = useRef<HTMLInputElement>(null);
   428→   428→
   429→   429→  function pick(f: File | null) {
   430→   430→    setError(null);
   431→   431→    setFile(null);
   432→   432→    setPreview(null);
   433→   433→    if (!f) return;
   434→   434→    if (!ALLOWED_IMAGE_TYPES.includes(f.type)) {
   435→   435→      setError("Only PNG, JPEG, WebP or GIF allowed.");
   436→   436→      return;
   437→   437→    }
   438→   438→    if (f.size > MAX_IMAGE_MB * 1024 * 1024) {

   446→   446→  async function submit() {
   447→   447→    setError(null);
   448→   448→    if (!supabaseConfigured) return;
   449→   449→    if (source === "upload" && !file) return;
   450→   450→    if (source === "giphy" && !gifPick) return;
   451→   451→    setState("sending");
   452→   452→    try {
   453→   453→      let uploadFile: File;

   455→   455→        setProgressMsg("Fetching GIF from GIPHY…");
   456→   456→        // Download the GIF into our own storage bucket — the dex never hotlinks

   462→   462→        setState("idle");
   463→   463→        return;
   464→   464→      }
   465→   465→
   466→   466→      setProgressMsg("Uploading…");
   467→   467→      const ext =
   468→   468→        uploadFile.type === "image/png"
   469→   469→          ? "png"
   470→   470→          : uploadFile.type === "image/webp"
   471→   471→            ? "webp"
   472→   472→            : uploadFile.type === "image/gif"
   473→   473→              ? "gif"
   474→   474→              : "jpg";
   475→   475→      const path = `pending/${mon.name}/${crypto.randomUUID()}.${ext}`;
   476→   476→      const { error: upErr } = await supabase.storage
   477→   477→        .from(MON_IMAGES_BUCKET)
   478→   478→        .upload(path, uploadFile, { contentType: uploadFile.type, cacheControl: "3600" });
   479→   479→      if (upErr) {
   480→   480→        setError(`Upload failed: ${upErr.message}`);
   481→   481→        setState("idle");
   482→   482→        return;
   483→   483→      }
   484→   484→
   485→   485→      setProgressMsg("Registering proposal…");
   486→   486→      const { error: dbErr } = await supabase.from("proposals").insert({
   487→   487→        mon_id: mon.id,
   488→   488→        kind: "image",
   489→   489→        content: path,
   490→   490→        submitted_by: canonicalize(nickname) || "anonymous",
   491→   491→      });
   492→   492→      if (dbErr) {
   493→   493→        await supabase.storage.from(MON_IMAGES_BUCKET).remove([path]);
   494→   494→        setError(
   495→   495→          /safety filter/i.test(dbErr.message)
   496→   496→            ? "Blocked by the safety filter."
   497→   497→            : `Could not register: ${dbErr.message}`
   498→   498→        );
   499→   499→        setState("idle");
   500→   500→        return;
   501→   501→      }
   502→   502→      setState("done");
   503→   503→    } catch (e) {
   504→   504→      setError(e instanceof Error ? e.message : "Network hiccup — try again.");
   505→   505→      setState("idle");
   506→   506→    } finally {
   507→   507→      setProgressMsg("");
   508→   508→    }
   509→   509→  }
   510→   510→
   511→   511→  if (state === "done") {
   512→   512→    return (
   513→   513→      <div className="space-y-3">
   514→   514→        <Alert className="border-pokedex-cyan/40 bg-pokedex-cyan/10">
   515→   515→          <ShieldCheck className="h-4 w-4 text-pokedex-cyan" />
   516→   516→          <AlertDescription className="font-soft text-sm">
   517→   517→            Artwork uploaded! It goes live after the channel team approves it.
   518→   518→          </AlertDescription>
   519→   519→        </Alert>
   520→   520→        <Button variant="secondary" className="font-soft rounded-full font-bold" onClick={() => setState("idle")}>
   521→   521→          Submit another
   522→   522→        </Button>
   523→   523→      </div>
   524→   524→    );
   525→   525→  }
   526→   526→
   527→   527→  return (
   528→   528→    <div className="space-y-4">
   529→   529→      <NicknameField nickname={nickname} setNickname={setNickname} />
   530→   530→      <div className="space-y-2">
   531→   531→        {GIPHY_ENABLED ? (

   533→   533→            <label className="font-soft text-sm font-bold text-muted-foreground">Artwork</label>

   538→   538→                  setSource("upload");
   539→   539→                  setGifPick(null);

   542→   542→                className={`font-soft flex items-center gap-1 rounded-full px-3 py-1 text-xs font-bold transition ${
   543→   543→                  source === "upload"
   544→   544→                    ? "bg-white text-primary shadow-[0_2px_8px_rgba(240,107,168,0.15)]"

   546→   546→                }`}
   547→   547→              >
   548→   548→                <UploadCloud className="h-3.5 w-3.5" /> Upload
   549→   549→              </button>
   550→   550→              <button
   551→   551→                type="button"

   562→   562→              >

   567→   567→        ) : (
   568→   568→          <label className="font-soft text-sm font-bold text-muted-foreground">Artwork (PNG/JPEG/WebP/GIF, max {MAX_IMAGE_MB}MB)</label>
   569→   569→        )}
   570→   570→        {source === "upload" ? (

   649→   649→/** Copy-to-clipboard share button producing a #mon-<name> deep link. */
   650→   650→function ShareButton({ mon }: { mon: Mon }) {
   651→   651→  const [copied, setCopied] = useState(false);
   652→   652→  const timer = useRef<ReturnType<typeof setTimeout> | null>(null);
   653→   653→
   654→   654→  useEffect(() => () => {
   655→   655→    if (timer.current) clearTimeout(timer.current);
   656→   656→  }, []);
   657→   657→
   658→   658→  async function copy() {
   659→   659→    const url = `${window.location.origin}/#mon-${mon.name}`;
   660→   660→    try {
   661→   661→      await navigator.clipboard.writeText(url);
   662→   662→    } catch {
   663→   663→      // clipboard API can be unavailable (http / older browsers)
   664→   664→      const ta = document.createElement("textarea");
   665→   665→      ta.value = url;
   666→   666→      ta.style.position = "fixed";
   667→   667→      ta.style.opacity = "0";
   668→   668→      document.body.appendChild(ta);
   669→   669→      ta.select();
   670→   670→      try {
   671→   671→        document.execCommand("copy");
   672→   672→      } catch {
   673→   673→        /* give up silently */
   674→   674→      }
   675→   675→      ta.remove();
   676→   676→    }
   677→   677→    setCopied(true);
   678→   678→    if (timer.current) clearTimeout(timer.current);
   679→   679→    timer.current = setTimeout(() => setCopied(false), 2000);
   680→   680→  }
   681→   681→
   682→   682→  return (
   683→   683→    <Button
   684→   684→      variant="secondary"
   685→   685→      size="sm"
   686→   686→      onClick={copy}
   687→   687→      className="font-soft shrink-0 gap-1.5 rounded-full border border-border px-3 font-bold"
   688→   688→      aria-label={`Copy share link for ${displayName(mon.name)}`}
   689→   689→    >
   690→   690→      {copied ? (
   691→   691→        <>
   692→   692→          <Check className="h-3.5 w-3.5 text-pokedex-cyan" />
   693→   693→          <span className="hidden sm:inline text-pokedex-cyan">COPIED!</span>
   694→   694→        </>
   695→   695→      ) : (
   696→   696→        <>
   697→   697→          <Link2 className="h-3.5 w-3.5" />
   698→   698→          <span className="hidden sm:inline">Share</span>
   699→   699→        </>
   700→   700→      )}
   701→   701→    </Button>
   702→   702→  );
   703→   703→}
