   367→   367→function DescriptionForm({
   368→   368→  mon,
   369→   369→  nickname,
   370→   370→  setNickname,
   371→   371→}: {
   372→   372→  mon: Mon;
   373→   373→  nickname: string;
   374→   374→  setNickname: (v: string) => void;
   375→   375→}) {
   376→   376→  const [text, setText] = useState("");
   377→   377→  const [state, setState] = useState<"idle" | "sending" | "done">("idle");
   378→   378→  const [error, setError] = useState<string | null>(null);
   379→   379→
   380→   380→  const clean = passesQuickFilter(text);
   381→   381→  const tooShort = text.trim().length < 12;
   382→   382→
   383→   383→  async function submit() {
   384→   384→    setError(null);
   385→   385→    if (!supabaseConfigured) {
   386→   386→      setError("Connection unavailable — try again soon.");
   387→   387→      return;
   388→   388→    }
   389→   389→    if (!clean) {
   390→   390→      setError("Your text didn't pass the safety filter — please rephrase.");
   391→   391→      return;
   392→   392→    }
   393→   393→    setState("sending");
   394→   394→    try {
   395→   395→      const { error: err } = await supabase.from("proposals").insert({
   396→   396→        mon_id: mon.id,
   397→   397→        kind: "description",
   398→   398→        content: text.trim().slice(0, 500),
   399→   399→        submitted_by: canonicalize(nickname) || "anonymous",
   400→   400→      });
   401→   401→      if (err) {
   402→   402→        const msg = /safety filter/i.test(err.message)
   403→   403→          ? "Blocked by the safety filter — please rephrase kindly."
   404→   404→          : err.message;
   405→   405→        setError(msg);
   406→   406→        setState("idle");
   407→   407→        return;
   408→   408→      }
   409→   409→      setState("done");
   410→   410→    } catch {
   411→   411→      setError("Network hiccup — try again.");
   412→   412→      setState("idle");
   413→   413→    }
   414→   414→  }
   415→   415→
   416→   416→  if (state === "done") {
   417→   417→    return (
   418→   418→      <div className="space-y-3">
   419→   419→        <Alert className="border-pokedex-cyan/40 bg-pokedex-cyan/10">
   420→   420→          <ShieldCheck className="h-4 w-4 text-pokedex-cyan" />
   421→   421→          <AlertDescription className="font-soft text-sm">
   422→   422→            Submitted! The channel team reviews every entry before it appears — thanks, researcher.
   423→   423→          </AlertDescription>
   424→   424→        </Alert>
   425→   425→        <Button variant="secondary" className="font-soft rounded-full font-bold" onClick={() => setState("idle")}>
   426→   426→          Propose another
   427→   427→        </Button>

   724→   724→/** Copy-to-clipboard share button producing a #mon-<name> deep link. */
   725→   725→function ShareButton({ mon }: { mon: Mon }) {
   726→   726→  const [copied, setCopied] = useState(false);
   727→   727→  const timer = useRef<ReturnType<typeof setTimeout> | null>(null);
   728→   728→
   729→   729→  useEffect(() => () => {
   730→   730→    if (timer.current) clearTimeout(timer.current);
   731→   731→  }, []);
   732→   732→
   733→   733→  async function copy() {
   734→   734→    const url = `${window.location.origin}/#mon-${mon.name}`;
   735→   735→    try {
   736→   736→      await navigator.clipboard.writeText(url);
   737→   737→    } catch {
   738→   738→      // clipboard API can be unavailable (http / older browsers)
   739→   739→      const ta = document.createElement("textarea");
   740→   740→      ta.value = url;
   741→   741→      ta.style.position = "fixed";
   742→   742→      ta.style.opacity = "0";
   743→   743→      document.body.appendChild(ta);
   744→   744→      ta.select();
   745→   745→      try {
   746→   746→        document.execCommand("copy");
   747→   747→      } catch {
   748→   748→        /* give up silently */
   749→   749→      }
   750→   750→      ta.remove();
   751→   751→    }
   752→   752→    setCopied(true);
   753→   753→    if (timer.current) clearTimeout(timer.current);
   754→   754→    timer.current = setTimeout(() => setCopied(false), 2000);
   755→   755→  }
   756→   756→
   757→   757→  return (
   758→   758→    <Button
   759→   759→      variant="secondary"
   760→   760→      size="sm"
   761→   761→      onClick={copy}
   762→   762→      className="font-soft shrink-0 gap-1.5 rounded-full border border-border px-3 font-bold"
   763→   763→      aria-label={`Copy share link for ${displayName(mon.name)}`}
   764→   764→    >
   765→   765→      {copied ? (
   766→   766→        <>
   767→   767→          <Check className="h-3.5 w-3.5 text-pokedex-cyan" />
   768→   768→          <span className="hidden sm:inline text-pokedex-cyan">COPIED!</span>
   769→   769→        </>
   770→   770→      ) : (
   771→   771→        <>
   772→   772→          <Link2 className="h-3.5 w-3.5" />
   773→   773→          <span className="hidden sm:inline">Share</span>
   774→   774→        </>
   775→   775→      )}
   776→   776→    </Button>
   777→   777→  );
   778→   778→}
