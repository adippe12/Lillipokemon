   178→  async function review(p: EnrichedProposal, approve: boolean) {

   185→        const { error: moveErr } = await supabase.storage.from(MON_IMAGES_BUCKET).move(p.content, newName);
   186→        if (moveErr) throw new Error(`Move failed: ${moveErr.message}`);
   187→        finalPath = newName;
   188→      }
   189→      const { error } = await supabase.rpc("review_proposal", {
   190→        p_proposal_id: p.id,
   191→        p_approve: approve,
   192→        p_final_image_path: finalPath,
   193→      });

   195→      if (!approve && p.kind === "image") {
   196→        await supabase.storage.from(MON_IMAGES_BUCKET).remove([p.content]).catch(() => {});
   197→      }
   198→      setMsg(approve ? "Approved and published." : "Rejected.");
   199→      await load();
   200→    } catch (e) {
   201→      setMsg(e instanceof Error ? e.message : "Action failed");

   203→      setBusyId(null);

   424→/**
   425→ * Full inline editor for one species: rename (moves the chat trigger word,
   426→ * storage folders and pending proposals), rewrite the story + credit, fix
   427→ * the discoverer and spotted count, and swap the approved artwork.
   428→ * The update_mon RPC treats NULL as "keep current value", so only changed
   429→ * fields are sent.
   430→ */
   431→function EditMonForm({
   432→  mon,
   433→  onDone,
   434→  onCancel,
   435→}: {
   436→  mon: Mon;
   437→  onDone: (msg: string) => void;
   438→  onCancel: () => void;
   439→}) {
   440→  const [name, setName] = useState(mon.name);
   441→  const [desc, setDesc] = useState(mon.description ?? "");
   442→  const [descBy, setDescBy] = useState(mon.description_by ?? "");
   443→  const [discBy, setDiscBy] = useState(mon.discovered_by);
   444→  const [count, setCount] = useState(String(mon.spotted_count));
   445→  const [file, setFile] = useState<File | null>(null);
   446→  const [preview, setPreview] = useState<string | null>(null);
   447→  const [busy, setBusy] = useState(false);
   448→  const [err, setErr] = useState<string | null>(null);
   449→  const fileRef = useRef<HTMLInputElement>(null);
   450→
   451→  const nextName = canonicalize(name) || mon.name;
   452→  const nameChanged = nextName !== mon.name;
   453→  const descChanged = desc.trim() !== (mon.description ?? "");
   454→  const descCleared = descChanged && desc.trim() === "";
   455→  const discChanged = discBy.trim() !== mon.discovered_by;
   456→  const countNum = Number.parseInt(count, 10);
   457→  const countChanged = Number.isFinite(countNum) && countNum !== mon.spotted_count;
   458→  const nothingChanged = !nameChanged && !descChanged && !discChanged && !countChanged && !file;
   459→
   460→  function pick(f: File | null) {
   461→    setErr(null);
   462→    setFile(null);
   463→    setPreview(null);
   464→    if (!f) return;
   465→    if (!ALLOWED_IMAGE_TYPES.includes(f.type)) {
   466→      setErr("Only PNG, JPEG, WebP or GIF allowed.");
   467→      return;
   468→    }
   469→    if (f.size > MAX_IMAGE_MB * 1024 * 1024) {
   470→      setErr(`Image too big — max ${MAX_IMAGE_MB}MB.`);
   471→      return;
   472→    }
   473→    setFile(f);
   474→    setPreview(URL.createObjectURL(f));
   475→  }
   476→
   477→  async function save() {
   478→    setErr(null);
   479→    if (desc.trim().length > 500) {
   480→      setErr("Description too long (max 500).");
   481→      return;
   482→    }
   483→    if (!Number.isFinite(countNum) || countNum < 0 || countNum > 1000000) {
   484→      setErr("Spotted count must be between 0 and 1,000,000.");
   485→      return;
   486→    }
   487→    if (nothingChanged) {
   488→      onCancel();
   489→      return;
   490→    }
   491→    setBusy(true);
   492→    let imagePath: string | null = null;
   493→    try {
   494→      if (file) {
   495→        const ext = file.type === "image/png" ? "png" : file.type === "image/webp" ? "webp" : file.type === "image/gif" ? "gif" : "jpg";
   496→        const path = `approved/${nextName}/${crypto.randomUUID()}.${ext}`;
   497→        const { error: upErr } = await supabase.storage
   498→          .from(MON_IMAGES_BUCKET)
   499→          .upload(path, file, { contentType: file.type, cacheControl: "3600" });
   500→        if (upErr) throw new Error(`Upload failed: ${upErr.message}`);
   501→        imagePath = path;
   502→      }
   503→      const { error } = await supabase.rpc("update_mon", {
   504→        p_mon_id: mon.id,
   505→        p_name: nameChanged ? nextName : null,
   506→        p_description: descChanged && !descCleared ? desc.trim() : null,
   507→        p_description_by: descChanged && !descCleared ? descBy.trim() : null,
   508→        p_clear_description: descCleared,
   509→        p_discovered_by: discChanged ? discBy.trim() : null,
   510→        p_spotted_count: countChanged ? countNum : null,
   511→        p_image_path: imagePath,
   512→        p_image_by: imagePath ? mon.image_by : null,
   513→        p_clear_image: false,
   514→      });
   515→      if (error) throw error;
   516→      // swap done — drop the previous approved file (best effort)
   517→      if (imagePath && mon.image_path) {
   518→        await supabase.storage.from(MON_IMAGES_BUCKET).remove([mon.image_path]).catch(() => {});
   519→      }
   520→      onDone(nameChanged ? `Renamed to ${displayName(nextName)} and updated.` : `${displayName(mon.name)} updated.`);
   521→    } catch (e) {
   522→      // roll back an already-uploaded file so a failed save leaves no orphans
   523→      if (imagePath) {
   524→        await supabase.storage.from(MON_IMAGES_BUCKET).remove([imagePath]).catch(() => {});
   525→      }
   526→      setErr(e instanceof Error ? e.message : "Update failed");
   527→    } finally {
   528→      setBusy(false);
   529→    }
   530→  }
   531→
   532→  return (
   533→    <div className="space-y-4 rounded-xl border border-primary/25 bg-secondary/40 p-4">
   534→      <div className="flex items-center justify-between">
   535→        <h4 className="font-display text-[11px] uppercase tracking-wide text-muted-foreground">
   536→          Edit {displayName(mon.name)} <span className="text-primary/60">#{mon.pokedex_no}</span>
   537→        </h4>
   538→        <button onClick={onCancel} className="font-soft text-xs font-bold text-muted-foreground transition hover:text-primary">
   539→          cancel
   540→        </button>
   541→      </div>
   542→
   543→      <div className="grid gap-4 sm:grid-cols-2">
   544→        <div className="space-y-1.5">
   545→          <Label className="font-soft text-xs">Species name (the chat trigger word)</Label>
   546→          <Input value={name} onChange={(e) => setName(e.target.value)} className="bg-background font-soft" />
   547→          <p className="font-soft text-[11px] leading-snug text-muted-foreground">
   548→            {nameChanged ? (
   549→              <>
   550→                renaming moves the chat word, artwork folders and pending proposals — becomes{" "}
   551→                <span className="font-bold text-primary">{nextName}</span>
   552→              </>
   553→            ) : (
   554→              <>lowercase word ending in &ldquo;mon&rdquo; — changing it changes what chat types to spot this species</>
   555→            )}
   556→          </p>
   557→        </div>
   558→
   559→        <div className="grid grid-cols-2 gap-3">
   560→          <div className="space-y-1.5">
   561→            <Label className="font-soft text-xs">Discovered by</Label>
   562→            <Input value={discBy} onChange={(e) => setDiscBy(e.target.value)} className="bg-background font-soft" />
   563→          </div>
   564→          <div className="space-y-1.5">
   565→            <Label className="font-soft text-xs">Spotted count</Label>
   566→            <Input type="number" min={0} value={count} onChange={(e) => setCount(e.target.value)} className="bg-background font-soft" />
   567→          </div>
   568→        </div>
   569→      </div>
   570→
   571→      <div className="space-y-1.5">
   572→        <div className="flex items-baseline justify-between">
   573→          <Label className="font-soft text-xs">Story / description</Label>
   574→          <span className="font-soft text-[11px] text-muted-foreground">
   575→            {desc.trim().length}/500 — saving it empty removes the story
   576→          </span>
   577→        </div>
   578→        <Textarea value={desc} onChange={(e) => setDesc(e.target.value)} maxLength={500} rows={3} className="bg-background font-soft" />
   579→        <Input
   580→          value={descBy}
   581→          onChange={(e) => setDescBy(e.target.value)}
   582→          placeholder="credit (who wrote it) — default: team"
   583→          className="bg-background font-soft text-xs"
   584→        />
   585→      </div>
   586→
   587→      <div className="space-y-1.5">
   588→        <Label className="font-soft text-xs">Artwork</Label>
   589→        <div className="flex flex-wrap items-center gap-3">
   590→          <div className="flex h-16 w-16 items-center justify-center overflow-hidden rounded-xl border border-border bg-background">
   591→            {preview ? (
   592→              <img src={preview} alt="New artwork preview" className="max-h-14 max-w-14 object-contain" />
   593→            ) : mon.image_path ? (
   594→              <img src={publicImageUrl(mon.image_path)} alt="" className="max-h-14 max-w-14 object-contain" />
   595→            ) : (
   596→              <MonSprite name={mon.name} seed={mon.id.slice(0, 8)} size={52} needsArt />
   597→            )}
   598→          </div>
   599→          <input
   600→            ref={fileRef}
   601→            type="file"
   602→            accept={ALLOWED_IMAGE_TYPES.join(",")}
   603→            className="hidden"
   604→            onChange={(e) => pick(e.target.files?.[0] ?? null)}
   605→          />
   606→          <Button size="sm" variant="outline" className="font-soft" onClick={() => fileRef.current?.click()} disabled={busy}>
   607→            <ImageIcon className="mr-1 h-3.5 w-3.5" />
   608→            {file ? "Change file" : mon.image_path ? "Replace artwork" : "Upload artwork"}
   609→          </Button>
   610→          {file && <span className="font-soft max-w-48 truncate text-xs text-muted-foreground">{file.name}</span>}
   611→          {file && (
   612→            <button onClick={() => pick(null)} className="font-soft text-xs font-bold text-muted-foreground transition hover:text-destructive">
   613→              undo
   614→            </button>
   615→          )}
   616→        </div>
   617→        <p className="font-soft text-[11px] leading-snug text-muted-foreground">
   618→          PNG / JPEG / WebP / GIF, max {MAX_IMAGE_MB}MB. Uploads land straight in the approved folder — public immediately.
   619→        </p>
   620→      </div>
   621→
   622→      {err && <p className="font-soft text-sm text-destructive">{err}</p>}
   623→
   624→      <div className="flex justify-end gap-2">
   625→        <Button size="sm" variant="ghost" className="font-soft" onClick={onCancel} disabled={busy}>
   626→          Cancel
   627→        </Button>
   628→        <Button size="sm" className="font-soft" onClick={() => void save()} disabled={busy || nothingChanged}>
   629→          {busy ? <Loader2 className="mr-1 h-4 w-4 animate-spin" /> : <Check className="mr-1 h-4 w-4" />} Save changes
   630→        </Button>
   631→      </div>
   632→    </div>
   633→  );
   634→}
