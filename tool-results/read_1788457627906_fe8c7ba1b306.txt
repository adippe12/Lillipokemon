   140→   140→function Console() {
   141→   141→  const [proposals, setProposals] = useState<EnrichedProposal[]>([]);
   142→   142→  const [mons, setMons] = useState<Mon[]>([]);

   144→   144→  const [msg, setMsg] = useState<string | null>(null);
   145→   145→  const [busyId, setBusyId] = useState<string | null>(null);
   146→   146→  const [removeTarget, setRemoveTarget] = useState<Mon | null>(null);
   147→   147→  const [removing, setRemoving] = useState(false);
   148→   148→  const [editId, setEditId] = useState<string | null>(null);
   149→   149→
   150→   150→  const load = useCallback(async () => {
   151→   151→    const [{ data: props }, { data: monsData }] = await Promise.all([
   152→   152→      supabase.from("proposals").select("*").order("created_at", { ascending: false }),
   153→   153→      supabase.from("mons").select("*").order("pokedex_no", { ascending: true }),
   154→   154→    ]);
   155→   155→    const monMap = new Map<string, Mon>((monsData ?? []).map((m) => [m.id, m as Mon]));
   156→   156→    setMons((monsData ?? []) as Mon[]);
   157→   157→    setProposals(
   158→   158→      ((props ?? []) as Proposal[]).map((p) => ({ ...p, mon: monMap.get(p.mon_id) }))
   159→   159→    );
   160→   160→    setLoading(false);
   161→   161→  }, []);
   162→   162→
   163→   163→  useEffect(() => {
   164→   164→    void load();
   165→   165→    const channel = supabase
   166→   166→      .channel("admin-live")
   167→   167→      .on("postgres_changes", { event: "*", schema: "public", table: "proposals" }, () => void load())
   168→   168→      .on("postgres_changes", { event: "*", schema: "public", table: "mons" }, () => void load())
   169→   169→      .subscribe();
   170→   170→    return () => {
   171→   171→      supabase.removeChannel(channel);
   172→   172→    };
   173→   173→  }, [load]);
   174→   174→
   175→   175→  const pendingDesc = useMemo(() => proposals.filter((p) => p.status === "pending" && p.kind === "description"), [proposals]);
   176→   176→  const pendingImg = useMemo(() => proposals.filter((p) => p.status === "pending" && p.kind === "image"), [proposals]);
   177→   177→
   178→   178→  async function review(p: EnrichedProposal, approve: boolean) {
   179→   179→    setBusyId(p.id);
   180→   180→    setMsg(null);
   181→   181→    try {
   182→   182→      let finalPath: string | null = null;
   183→   183→      if (approve && p.kind === "image" && p.mon) {
   184→   184→        const newName = `approved/${p.mon.name}/${p.id}.${p.content.split(".").pop() ?? "png"}`;
   185→   185→        const { error: moveErr } = await supabase.storage.from(MON_IMAGES_BUCKET).move(p.content, newName);
   186→   186→        if (moveErr) throw new Error(`Move failed: ${moveErr.message}`);
   187→   187→        finalPath = newName;
   188→   188→      }
   189→   189→      const { error } = await supabase.rpc("review_proposal", {
   190→   190→        p_proposal_id: p.id,
   191→   191→        p_approve: approve,
   192→   192→        p_final_image_path: finalPath,
   193→   193→      });
   194→   194→      if (error) throw error;
   195→   195→      if (!approve && p.kind === "image") {
   196→   196→        await supabase.storage.from(MON_IMAGES_BUCKET).remove([p.content]).catch(() => {});
   197→   197→      }
   198→   198→      setMsg(approve ? "Approved and published." : "Rejected.");
   199→   199→      await load();
   200→   200→    } catch (e) {
   201→   201→      setMsg(e instanceof Error ? e.message : "Action failed");
   202→   202→    } finally {
   203→   203→      setBusyId(null);
   204→   204→    }
   205→   205→  }
   206→   206→
   207→   207→  async function clearField(mon: Mon, field: "description" | "image") {
   208→   208→    setBusyId(mon.id);
   209→   209→    const { error } = await supabase.rpc("clear_mon_field", { p_mon_id: mon.id, p_field: field });
   210→   210→    setMsg(error ? error.message : `${field} cleared.`);
   211→   211→    await load();
   212→   212→    setBusyId(null);
   213→   213→  }
   214→   214→
   215→   215→  async function removeMon(mon: Mon) {
   216→   216→    setRemoving(true);
   217→   217→    setMsg(null);

   219→   219→      // delete known image folders first (console has storage manage rights)

   222→   222→        const paths = (objs ?? []).map((o) => `${prefix}/${o.name}`);
   223→   223→        if (paths.length) {
   224→   224→          await supabase.storage.from(MON_IMAGES_BUCKET).remove(paths).catch(() => {});
   225→   225→        }
   226→   226→      }
   227→   227→      const { error } = await supabase.rpc("delete_mon", { p_mon_id: mon.id });
   228→   228→      if (error) throw error;
   229→   229→      setMsg(`${displayName(mon.name)} removed — entry, proposals, images and its trigger word are gone.`);
   230→   230→      setRemoveTarget(null);
   231→   231→      await load();
   232→   232→    } catch (e) {
   233→   233→      setMsg(e instanceof Error ? e.message : "Remove failed");
   234→   234→    } finally {
   235→   235→      setRemoving(false);
   236→   236→    }
   237→   237→  }
   238→   238→
   239→   239→  return (
   240→   240→    <div className="space-y-6">
   241→   241→      {msg && (

   266→   266→        </TabsList>

   268→   268→        <TabsContent value="descriptions" className="mt-4 space-y-3">

   271→   271→          ) : pendingDesc.length === 0 ? (
   272→   272→            <Empty text="No descriptions waiting for review." />
   273→   273→          ) : (
   274→   274→            pendingDesc.map((p) => (
   275→   275→              <div key={p.id} className="flex flex-col gap-3 rounded-xl border border-border bg-card p-4 sm:flex-row sm:items-center">
   276→   276→                <div className="min-w-0 flex-1">

   280→   280→                  </div>
   281→   281→                  <p className="text-sm text-foreground">&ldquo;{p.content}&rdquo;</p>
   282→   282→                </div>

   286→   286→                  </Button>

   288→   288→                    <X className="mr-1 h-4 w-4" /> Reject

   424→   424→/**
   425→   425→ * Full inline editor for one species: rename (moves the chat trigger word,
   426→   426→ * storage folders and pending proposals), rewrite the story + credit, fix
   427→   427→ * the discoverer and spotted count, and swap the approved artwork.
   428→   428→ * The update_mon RPC treats NULL as "keep current value", so only changed
   429→   429→ * fields are sent.
   430→   430→ */
   431→   431→function EditMonForm({
   432→   432→  mon,
   433→   433→  onDone,
   434→   434→  onCancel,
   435→   435→}: {
   436→   436→  mon: Mon;
   437→   437→  onDone: (msg: string) => void;
   438→   438→  onCancel: () => void;
   439→   439→}) {
   440→   440→  const [name, setName] = useState(mon.name);
   441→   441→  const [desc, setDesc] = useState(mon.description ?? "");
   442→   442→  const [descBy, setDescBy] = useState(mon.description_by ?? "");
   443→   443→  const [discBy, setDiscBy] = useState(mon.discovered_by);
   444→   444→  const [count, setCount] = useState(String(mon.spotted_count));
   445→   445→  const [file, setFile] = useState<File | null>(null);
   446→   446→  const [preview, setPreview] = useState<string | null>(null);
   447→   447→  const [busy, setBusy] = useState(false);
   448→   448→  const [err, setErr] = useState<string | null>(null);
   449→   449→  const fileRef = useRef<HTMLInputElement>(null);
   450→   450→
   451→   451→  const nextName = canonicalize(name) || mon.name;
   452→   452→  const nameChanged = nextName !== mon.name;
   453→   453→  const descChanged = desc.trim() !== (mon.description ?? "");
   454→   454→  const descCleared = descChanged && desc.trim() === "";
   455→   455→  const discChanged = discBy.trim() !== mon.discovered_by;
   456→   456→  const countNum = Number.parseInt(count, 10);
   457→   457→  const countChanged = Number.isFinite(countNum) && countNum !== mon.spotted_count;
   458→   458→  const nothingChanged = !nameChanged && !descChanged && !discChanged && !countChanged && !file;
   459→   459→
   460→   460→  function pick(f: File | null) {
   461→   461→    setErr(null);
   462→   462→    setFile(null);
   463→   463→    setPreview(null);
   464→   464→    if (!f) return;
   465→   465→    if (!ALLOWED_IMAGE_TYPES.includes(f.type)) {
   466→   466→      setErr("Only PNG, JPEG, WebP or GIF allowed.");
   467→   467→      return;
   468→   468→    }
   469→   469→    if (f.size > MAX_IMAGE_MB * 1024 * 1024) {
   470→   470→      setErr(`Image too big — max ${MAX_IMAGE_MB}MB.`);
   471→   471→      return;
   472→   472→    }
   473→   473→    setFile(f);
   474→   474→    setPreview(URL.createObjectURL(f));
   475→   475→  }
   476→   476→
   477→   477→  async function save() {
   478→   478→    setErr(null);
   479→   479→    if (desc.trim().length > 500) {
   480→   480→      setErr("Description too long (max 500).");
   481→   481→      return;
   482→   482→    }
   483→   483→    if (!Number.isFinite(countNum) || countNum < 0 || countNum > 1000000) {
   484→   484→      setErr("Spotted count must be between 0 and 1,000,000.");
   485→   485→      return;
   486→   486→    }
   487→   487→    if (nothingChanged) {
   488→   488→      onCancel();
   489→   489→      return;
   490→   490→    }
   491→   491→    setBusy(true);
   492→   492→    let imagePath: string | null = null;
   493→   493→    try {
   494→   494→      if (file) {
   495→   495→        const ext = file.type === "image/png" ? "png" : file.type === "image/webp" ? "webp" : file.type === "image/gif" ? "gif" : "jpg";
   496→   496→        const path = `approved/${nextName}/${crypto.randomUUID()}.${ext}`;
   497→   497→        const { error: upErr } = await supabase.storage
   498→   498→          .from(MON_IMAGES_BUCKET)
   499→   499→          .upload(path, file, { contentType: file.type, cacheControl: "3600" });
   500→   500→        if (upErr) throw new Error(`Upload failed: ${upErr.message}`);
   501→   501→        imagePath = path;
   502→   502→      }
   503→   503→      const { error } = await supabase.rpc("update_mon", {
   504→   504→        p_mon_id: mon.id,
   505→   505→        p_name: nameChanged ? nextName : null,
   506→   506→        p_description: descChanged && !descCleared ? desc.trim() : null,
   507→   507→        p_description_by: descChanged && !descCleared ? descBy.trim() : null,
   508→   508→        p_clear_description: descCleared,
   509→   509→        p_discovered_by: discChanged ? discBy.trim() : null,
   510→   510→        p_spotted_count: countChanged ? countNum : null,
   511→   511→        p_image_path: imagePath,
   512→   512→        p_image_by: imagePath ? mon.image_by : null,
   513→   513→        p_clear_image: false,
   514→   514→      });
   515→   515→      if (error) throw error;
   516→   516→      // swap done — drop the previous approved file (best effort)
   517→   517→      if (imagePath && mon.image_path) {
   518→   518→        await supabase.storage.from(MON_IMAGES_BUCKET).remove([mon.image_path]).catch(() => {});
   519→   519→      }
   520→   520→      onDone(nameChanged ? `Renamed to ${displayName(nextName)} and updated.` : `${displayName(mon.name)} updated.`);
   521→   521→    } catch (e) {
   522→   522→      // roll back an already-uploaded file so a failed save leaves no orphans
   523→   523→      if (imagePath) {
   524→   524→        await supabase.storage.from(MON_IMAGES_BUCKET).remove([imagePath]).catch(() => {});
   525→   525→      }
   526→   526→      setErr(e instanceof Error ? e.message : "Update failed");
   527→   527→    } finally {
   528→   528→      setBusy(false);
   529→   529→    }
   530→   530→  }
   531→   531→
   532→   532→  return (
   533→   533→    <div className="space-y-4 rounded-xl border border-primary/25 bg-secondary/40 p-4">
   534→   534→      <div className="flex items-center justify-between">
   535→   535→        <h4 className="font-display text-[11px] uppercase tracking-wide text-muted-foreground">
   536→   536→          Edit {displayName(mon.name)} <span className="text-primary/60">#{mon.pokedex_no}</span>
   537→   537→        </h4>
   538→   538→        <button onClick={onCancel} className="font-soft text-xs font-bold text-muted-foreground transition hover:text-primary">
   539→   539→          cancel
   540→   540→        </button>
   541→   541→      </div>
   542→   542→
   543→   543→      <div className="grid gap-4 sm:grid-cols-2">
   544→   544→        <div className="space-y-1.5">
   545→   545→          <Label className="font-soft text-xs">Species name (the chat trigger word)</Label>
   546→   546→          <Input value={name} onChange={(e) => setName(e.target.value)} className="bg-background font-soft" />
   547→   547→          <p className="font-soft text-[11px] leading-snug text-muted-foreground">
   548→   548→            {nameChanged ? (
   549→   549→              <>
   550→   550→                renaming moves the chat word, artwork folders and pending proposals — becomes{" "}
   551→   551→                <span className="font-bold text-primary">{nextName}</span>
   552→   552→              </>
   553→   553→            ) : (
   554→   554→              <>lowercase word ending in &ldquo;mon&rdquo; — changing it changes what chat types to spot this species</>
   555→   555→            )}
   556→   556→          </p>
   557→   557→        </div>
   558→   558→
   559→   559→        <div className="grid grid-cols-2 gap-3">
   560→   560→          <div className="space-y-1.5">
   561→   561→            <Label className="font-soft text-xs">Discovered by</Label>
   562→   562→            <Input value={discBy} onChange={(e) => setDiscBy(e.target.value)} className="bg-background font-soft" />
   563→   563→          </div>
   564→   564→          <div className="space-y-1.5">
   565→   565→            <Label className="font-soft text-xs">Spotted count</Label>
   566→   566→            <Input type="number" min={0} value={count} onChange={(e) => setCount(e.target.value)} className="bg-background font-soft" />
   567→   567→          </div>
   568→   568→        </div>
   569→   569→      </div>
   570→   570→
   571→   571→      <div className="space-y-1.5">
   572→   572→        <div className="flex items-baseline justify-between">
   573→   573→          <Label className="font-soft text-xs">Story / description</Label>
   574→   574→          <span className="font-soft text-[11px] text-muted-foreground">
   575→   575→            {desc.trim().length}/500 — saving it empty removes the story
   576→   576→          </span>
   577→   577→        </div>
   578→   578→        <Textarea value={desc} onChange={(e) => setDesc(e.target.value)} maxLength={500} rows={3} className="bg-background font-soft" />
   579→   579→        <Input
   580→   580→          value={descBy}
   581→   581→          onChange={(e) => setDescBy(e.target.value)}
   582→   582→          placeholder="credit (who wrote it) — default: team"
   583→   583→          className="bg-background font-soft text-xs"
   584→   584→        />
   585→   585→      </div>

   587→   587→      <div className="space-y-1.5">
   588→   588→        <Label className="font-soft text-xs">Artwork</Label>

   592→   592→              <img src={preview} alt="New artwork preview" className="max-h-14 max-w-14 object-contain" />
