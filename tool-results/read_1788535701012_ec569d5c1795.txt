    46→function blip(kind: "new" | "spot") {
    47→  try {
    48→    const Ctx = window.AudioContext ?? (window as unknown as { webkitAudioContext?: typeof AudioContext }).webkitAudioContext;
    49→    if (!Ctx) return;
    50→    const ctx = new Ctx();
    51→    const now = ctx.currentTime;
    52→    const notes = kind === "new" ? [523.25, 659.25, 783.99] : [440, 587.33];
    53→    notes.forEach((f, i) => {
    54→      const o = ctx.createOscillator();
    55→      const g = ctx.createGain();
    56→      o.type = "sine";
    57→      o.frequency.value = f;
    58→      g.gain.setValueAtTime(0.0001, now + i * 0.09);
    59→      g.gain.exponentialRampToValueAtTime(0.03, now + i * 0.09 + 0.02);
    60→      g.gain.exponentialRampToValueAtTime(0.0001, now + i * 0.09 + 0.16);
    61→      o.connect(g).connect(ctx.destination);
    62→      o.start(now + i * 0.09);
    63→      o.stop(now + i * 0.09 + 0.16);
    64→    });
    65→    setTimeout(() => ctx.close().catch(() => {}), 900);
    66→  } catch {
    67→    /* audio not available */
    68→  }
    69→}

    95→function useTypewriter(words: string[]): string {
    96→  const [typed, setTyped] = useState("");
    97→  useEffect(() => {
    98→    if (window.matchMedia?.("(prefers-reduced-motion: reduce)").matches) {
    99→      const t = setTimeout(() => setTyped(words[0]), 0);
   100→      return () => clearTimeout(t);
   101→    }
   102→    let i = 0;
   103→    let pos = 0;
   104→    let dir = 1;
   105→    let timer: ReturnType<typeof setTimeout>;
   106→    const step = () => {
   107→      const word = words[i % words.length];
   108→      pos += dir;
   109→      setTyped(word.slice(0, Math.max(0, pos)));
   110→      let delay = dir > 0 ? 110 : 40;
   111→      if (dir > 0 && pos >= word.length) {
   112→        dir = -1;
   113→        delay = 1500;
   114→      } else if (dir < 0 && pos <= 0) {
   115→        dir = 1;
   116→        i += 1;
   117→        delay = 400;
   118→      }
   119→      timer = setTimeout(step, delay);
   120→    };
   121→    timer = setTimeout(step, 700);
   122→    return () => clearTimeout(timer);
   123→  }, [words]);
   124→  return typed;
   125→}

   127→// ---- animated number (count-up on change, reduced-motion aware) ----
   128→function useCountUp(target: number, duration = 700): number {
   129→  const [val, setVal] = useState(target);
   130→  const prevRef = useRef(target);
   131→  useEffect(() => {
   132→    const from = prevRef.current;
   133→    prevRef.current = target;
   134→    if (from === target) return;
   135→    if (window.matchMedia?.("(prefers-reduced-motion: reduce)").matches) {
   136→      const raf = requestAnimationFrame(() => setVal(target));
   137→      return () => cancelAnimationFrame(raf);
   138→    }
   139→    const t0 = performance.now();
   140→    let raf = 0;
   141→    const tick = (t: number) => {
   142→      const p = Math.min(1, (t - t0) / duration);
   143→      const eased = 1 - Math.pow(1 - p, 3);
   144→      setVal(Math.round(from + (target - from) * eased));
   145→      if (p < 1) raf = requestAnimationFrame(tick);
   146→    };
   147→    raf = requestAnimationFrame(tick);
   148→    return () => cancelAnimationFrame(raf);
   149→  }, [target, duration]);
   150→  return val;
   151→}
