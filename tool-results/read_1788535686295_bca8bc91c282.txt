   292→function DescriptionForm({
   293→  mon,
   294→  nickname,
   295→  setNickname,
   296→}: {
   297→  mon: Mon;
   298→  nickname: string;
   299→  setNickname: (v: string) => void;
   300→}) {
   301→  const [text, setText] = useState("");
   302→  const [state, setState] = useState<"idle" | "sending" | "done">("idle");
   303→  const [error, setError] = useState<string | null>(null);
   304→
   305→  const clean = passesQuickFilter(text);
   306→  const tooShort = text.trim().length < 12;
   307→
   308→  async function submit() {
   309→    setError(null);
   310→    if (!supabaseConfigured) {
   311→      setError("Connection unavailable — try again soon.");
   312→      return;
   313→    }
   314→    if (!clean) {
   315→      setError("Your text didn't pass the safety filter — please rephrase.");
   316→      return;
   317→    }
   318→    setState("sending");
   319→    try {
   320→      const { error: err } = await supabase.from("proposals").insert({
   321→        mon_id: mon.id,
   322→        kind: "description",
   323→        content: text.trim().slice(0, 500),
   324→        submitted_by: canonicalize(nickname) || "anonymous",
   325→      });
   326→      if (err) {
   327→        const msg = /safety filter/i.test(err.message)
   328→          ? "Blocked by the safety filter — please rephrase kindly."
   329→          : err.message;
   330→        setError(msg);
   331→        setState("idle");
   332→        return;
   333→      }
   334→      setState("done");
   335→    } catch {
   336→      setError("Network hiccup — try again.");
   337→      setState("idle");
   338→    }
   339→  }
   340→
   341→  if (state === "done") {
   342→    return (
   343→      <div className="space-y-3">
   344→        <Alert className="border-pokedex-cyan/40 bg-pokedex-cyan/10">
   345→          <ShieldCheck className="h-4 w-4 text-pokedex-cyan" />
   346→          <AlertDescription className="font-soft text-sm">
   347→            Submitted! The channel team reviews every entry before it appears — thanks, researcher.
   348→          </AlertDescription>
   349→        </Alert>
   350→        <Button variant="secondary" className="font-soft rounded-full font-bold" onClick={() => setState("idle")}>
   351→          Propose another
   352→        </Button>
   353→      </div>
   354→    );
   355→  }
   356→
   357→  return (
   358→    <div className="space-y-4">
   359→      <NicknameField nickname={nickname} setNickname={setNickname} />
   360→      <div className="space-y-1.5">
   361→        <label htmlFor="lp-desc" className="font-soft text-sm font-bold text-muted-foreground">
   362→          What is {displayName(mon.name)}? (max {MAX_DESCRIPTION} chars)
   363→        </label>
   364→        <Textarea
   365→          id="lp-desc"
   366→          value={text}
   367→          maxLength={MAX_DESCRIPTION}
   368→          onChange={(e) => setText(e.target.value)}
   369→          placeholder="A gremlin of pure chaos that only appears when someone drops a combo..."
   370→          rows={4}
   371→          className="resize-none rounded-2xl bg-secondary/70 font-semibold"
   372→        />
   373→        <div className="flex items-center gap-3">
   374→          <Progress value={(text.length / MAX_DESCRIPTION) * 100} className="h-1.5" />
   375→          <span className="font-soft shrink-0 text-xs text-muted-foreground">
   376→            {text.length}/{MAX_DESCRIPTION}
   377→          </span>
   378→        </div>
   379→        {!clean && (
   380→          <p className="font-soft text-xs text-destructive">
   381→            That wording will be blocked by the safety filter — please rephrase kindly.
   382→          </p>
   383→        )}
   384→      </div>
   385→      {error && <p className="font-soft text-sm text-destructive">{error}</p>}
   386→      <Button
   387→        onClick={submit}
   388→        disabled={state === "sending" || tooShort || !clean}
   389→        className="font-display w-full rounded-full text-sm font-bold shadow-[0_6px_18px_rgba(240,107,168,0.35)] hover:bg-pokedex-dark-red"
   390→      >
   391→        {state === "sending" ? (
   392→          <>
   393→            <Loader2 className="mr-2 h-4 w-4 animate-spin" /> Sending…
   394→          </>
   395→        ) : (
   396→          <>
   397→            <Sparkles className="mr-2 h-4 w-4" /> Submit for review
   398→          </>
   399→        )}
   400→      </Button>
   401→      <p className="flex items-center gap-1.5 text-xs text-muted-foreground">
   402→        <ShieldCheck className="h-3.5 w-3.5 text-pokedex-cyan" />
   403→        Safety: filtered for bad words + human-approved by the channel team before publishing.
   404→      </p>
   405→    </div>
   406→  );
   407→}

   409→function ImageForm({
   410→  mon,
   411→  nickname,
   412→  setNickname,
   413→}: {
   414→  mon: Mon;
   415→  nickname: string;
   416→  setNickname: (v: string) => void;
   417→}) {
   418→  const [file, setFile] = useState<File | null>(null);
   419→  const [preview, setPreview] = useState<string | null>(null);
   420→  // Artwork source: a hand-picked file, or one found via the GIPHY picker.
   421→  // The GIPHY option only renders when a free API key is configured.
   422→  const [source, setSource] = useState<"upload" | "giphy">("upload");
   423→  const [gifPick, setGifPick] = useState<GiphyGif | null>(null);
   424→  const [state, setState] = useState<"idle" | "sending" | "done">("idle");
   425→  const [progressMsg, setProgressMsg] = useState("");
   426→  const [error, setError] = useState<string | null>(null);
   427→  const inputRef = useRef<HTMLInputElement>(null);
   428→
   429→  function pick(f: File | null) {
   430→    setError(null);
   431→    setFile(null);
   432→    setPreview(null);
   433→    if (!f) return;
   434→    if (!ALLOWED_IMAGE_TYPES.includes(f.type)) {
   435→      setError("Only PNG, JPEG, WebP or GIF allowed.");
   436→      return;
   437→    }
   438→    if (f.size > MAX_IMAGE_MB * 1024 * 1024) {
   439→      setError(`Image too big — max ${MAX_IMAGE_MB}MB.`);
   440→      return;
   441→    }
   442→    setFile(f);
   443→    setPreview(URL.createObjectURL(f));
   444→  }
   445→
   446→  async function submit() {
   447→    setError(null);
   448→    if (!supabaseConfigured) return;
   449→    if (source === "upload" && !file) return;
   450→    if (source === "giphy" && !gifPick) return;
   451→    setState("sending");
   452→    try {
   453→      let uploadFile: File;
   454→      if (source === "giphy" && gifPick) {
   455→        setProgressMsg("Fetching GIF from GIPHY…");
   456→        // Download the GIF into our own storage bucket — the dex never hotlinks
   457→        // GIPHY, so approval + hosting + rendering stay exactly the same.
   458→        uploadFile = await fetchGifAsFile(gifPick, mon.name);
   459→      } else if (file) {
   460→        uploadFile = file;
   461→      } else {
   462→        setState("idle");
   463→        return;
   464→      }
   465→
   466→      setProgressMsg("Uploading…");
   467→      const ext =
   468→        uploadFile.type === "image/png"
   469→          ? "png"
   470→          : uploadFile.type === "image/webp"
   471→            ? "webp"
   472→            : uploadFile.type === "image/gif"
   473→              ? "gif"
   474→              : "jpg";
   475→      const path = `pending/${mon.name}/${crypto.randomUUID()}.${ext}`;
   476→      const { error: upErr } = await supabase.storage
   477→        .from(MON_IMAGES_BUCKET)
   478→        .upload(path, uploadFile, { contentType: uploadFile.type, cacheControl: "3600" });
   479→      if (upErr) {
   480→        setError(`Upload failed: ${upErr.message}`);
   481→        setState("idle");
   482→        return;
   483→      }
   484→
   485→      setProgressMsg("Registering proposal…");
   486→      const { error: dbErr } = await supabase.from("proposals").insert({
   487→        mon_id: mon.id,
   488→        kind: "image",
   489→        content: path,
   490→        submitted_by: canonicalize(nickname) || "anonymous",
   491→      });
   492→      if (dbErr) {
   493→        await supabase.storage.from(MON_IMAGES_BUCKET).remove([path]);
   494→        setError(
   495→          /safety filter/i.test(dbErr.message)
   496→            ? "Blocked by the safety filter."
   497→            : `Could not register: ${dbErr.message}`
   498→        );
   499→        setState("idle");
   500→        return;
   501→      }
   502→      setState("done");
   503→    } catch (e) {
   504→      setError(e instanceof Error ? e.message : "Network hiccup — try again.");
   505→      setState("idle");
   506→    } finally {
   507→      setProgressMsg("");
   508→    }
   509→  }
   510→
   511→  if (state === "done") {
   512→    return (
   513→      <div className="space-y-3">
   514→        <Alert className="border-pokedex-cyan/40 bg-pokedex-cyan/10">
   515→          <ShieldCheck className="h-4 w-4 text-pokedex-cyan" />
   516→          <AlertDescription className="font-soft text-sm">
   517→            Artwork uploaded! It goes live after the channel team approves it.
   518→          </AlertDescription>
   519→        </Alert>
   520→        <Button variant="secondary" className="font-soft rounded-full font-bold" onClick={() => setState("idle")}>
   521→          Submit another
   522→        </Button>
   523→      </div>
   524→    );
   525→  }
   526→
   527→  return (
   528→    <div className="space-y-4">
   529→      <NicknameField nickname={nickname} setNickname={setNickname} />
   530→      <div className="space-y-2">

   649→/** Copy-to-clipboard share button producing a #mon-<name> deep link. */
   650→function ShareButton({ mon }: { mon: Mon }) {
   651→  const [copied, setCopied] = useState(false);
   652→  const timer = useRef<ReturnType<typeof setTimeout> | null>(null);
   653→
   654→  useEffect(() => () => {
   655→    if (timer.current) clearTimeout(timer.current);
   656→  }, []);
   657→
   658→  async function copy() {
   659→    const url = `${window.location.origin}/#mon-${mon.name}`;
   660→    try {
   661→      await navigator.clipboard.writeText(url);
   662→    } catch {
   663→      // clipboard API can be unavailable (http / older browsers)
   664→      const ta = document.createElement("textarea");
   665→      ta.value = url;
   666→      ta.style.position = "fixed";
   667→      ta.style.opacity = "0";
   668→      document.body.appendChild(ta);
   669→      ta.select();
   670→      try {
   671→        document.execCommand("copy");
   672→      } catch {
   673→        /* give up silently */
   674→      }
   675→      ta.remove();
   676→    }
   677→    setCopied(true);
   678→    if (timer.current) clearTimeout(timer.current);
   679→    timer.current = setTimeout(() => setCopied(false), 2000);
   680→  }
   681→
   682→  return (
   683→    <Button
   684→      variant="secondary"
   685→      size="sm"
   686→      onClick={copy}
   687→      className="font-soft shrink-0 gap-1.5 rounded-full border border-border px-3 font-bold"
   688→      aria-label={`Copy share link for ${displayName(mon.name)}`}
   689→    >
   690→      {copied ? (
   691→        <>
   692→          <Check className="h-3.5 w-3.5 text-pokedex-cyan" />
   693→          <span className="hidden sm:inline text-pokedex-cyan">COPIED!</span>
   694→        </>
   695→      ) : (
   696→        <>
   697→          <Link2 className="h-3.5 w-3.5" />
   698→          <span className="hidden sm:inline">Share</span>
   699→        </>
   700→      )}
   701→    </Button>
   702→  );
   703→}

   705→/** Horizontal bar comparing this species' spot count to the most-spotted one. */
   706→function PopularityBar({ spotted, maxSpotted }: { spotted: number; maxSpotted: number }) {
   707→  const pct = Math.max(4, Math.round((spotted / maxSpotted) * 100));
   708→  const isTop = spotted >= maxSpotted;

   710→    <div className="rounded-2xl border border-border bg-secondary/60 px-4 py-3">
   711→      <div className="mb-1.5 flex items-center justify-between font-soft text-xs font-bold">
   712→        <span className="flex items-center gap-1 text-muted-foreground">
   713→          <Flame className="h-3.5 w-3.5 text-primary" aria-hidden />
   714→          DEX POPULARITY · {formatNumber(spotted)} spots
   715→        </span>
   716→        <span className={cn("flex items-center gap-1", isTop ? "text-pokedex-yellow" : "text-muted-foreground")}>
   717→          {isTop && <Crown className="h-3.5 w-3.5" aria-hidden />}
   718→          {isTop ? "top species!" : `${Math.round((spotted / maxSpotted) * 100)}% of top record`}
   719→        </span>
   720→      </div>
   721→      <div className="h-2.5 overflow-hidden rounded-full bg-white shadow-[inset_0_1px_3px_rgba(240,107,168,0.15)]">

   723→          className="h-full rounded-full bg-gradient-to-r from-primary to-[#b9a7f2] transition-[width] duration-700"

   725→          role="meter"
   726→          aria-valuenow={spotted}
   727→          aria-valuemin={0}
   728→          aria-valuemax={maxSpotted}
   729→          aria-label="Popularity compared to the most spotted species"
   730→        />
   731→      </div>
   732→    </div>
